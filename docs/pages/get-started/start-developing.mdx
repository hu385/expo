title: Traveling App

import React, { useState } from "react";
import { View, Text, Button, Image, FlatList } from "react-native";
import MapboxGL from "@rnmapbox/maps";
import * as ImagePicker from "expo-image-picker";

MapboxGL.setAccessToken("DIN_MAPBOX_TOKEN");

export default function App() {
  const [selectedCountry, setSelectedCountry] = useState(null);
  const [photos, setPhotos] = useState({}); // { countryCode: [uri, uri] }

  const pickImage = async () => {
    let result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      allowsMultipleSelection: false,
    });

    if (!result.canceled) {
      setPhotos((prev) => ({
        ...prev,
        [selectedCountry]: [
          ...(prev[selectedCountry] || []),
          result.assets[0].uri,
        ].slice(0, 5), // max 5 bilder
      }));
    }
  };

  return (
    <View style={{ flex: 1 }}>
      <MapboxGL.MapView style={{ flex: 1 }}>
        <MapboxGL.Camera zoomLevel={1.5} centerCoordinate={[0, 20]} />
        {/* Här kan du lägga polygoner för länder senare */}
      </MapboxGL.MapView>

      {selectedCountry && (
        <View style={{ position: "absolute", bottom: 20, left: 20, right: 20, backgroundColor: "white", padding: 10 }}>
          <Text>{selectedCountry}</Text>
          <Button title="Lägg till bild" onPress={pickImage} />

          <FlatList
            horizontal
            data={photos[selectedCountry] || []}
            renderItem={({ item }) => <Image source={{ uri: item }} style={{ width: 80, height: 80, margin: 5 }} />}
            keyExtractor={(item, index) => index.toString()}
          />
        </View>
      )}
    </View>
  );
}
